cmake_minimum_required(VERSION 3.24)
project(OpenFilter VERSION 0.3.3 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(CTest)
include(cmake/Dependencies.cmake)

option(OPENFILTER_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
add_library(openfilter_options INTERFACE)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(openfilter_options INTERFACE -Wall -Wextra -Wpedantic -Werror)
  if(OPENFILTER_SANITIZE)
    target_compile_options(openfilter_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(openfilter_options INTERFACE -fsanitize=address,undefined)
  endif()
endif()
add_subdirectory(libs)
add_subdirectory(plugins/eq)
add_subdirectory(plugins/compressor)
add_subdirectory(plugins/limiter)
add_subdirectory(plugins/reverb)
install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION share/openfilter)

add_executable(eq_render tools/eq_render.cpp)
target_link_libraries(eq_render PRIVATE openfilter_eq_engine openfilter_options)
add_executable(editor_preview tools/editor_preview.cpp)
target_link_libraries(editor_preview PRIVATE openfilter_eq_editor openfilter_options)
if(BUILD_TESTING)
  find_package(Threads REQUIRED)
  add_executable(dsp_tests tests/dsp_tests.cpp)
  target_link_libraries(dsp_tests PRIVATE openfilter_eq_engine openfilter_options)
  add_test(NAME dsp COMMAND dsp_tests)
  add_executable(clap_tests tests/clap_tests.cpp tests/AllocationGuard.cpp)
  target_link_libraries(clap_tests PRIVATE clap openfilter_eq_engine openfilter_plugin openfilter_options Threads::Threads PkgConfig::CAIRO ${CMAKE_DL_LIBS})
  add_test(NAME clap_contract COMMAND clap_tests $<TARGET_FILE:OpenFilterEQ>)
  add_executable(ui_tests tests/ui_tests.cpp)
  target_link_libraries(ui_tests PRIVATE openfilter_eq_editor openfilter_options)
  add_test(NAME ui COMMAND ui_tests)
  pkg_check_modules(FONTCONFIG REQUIRED IMPORTED_TARGET fontconfig)
  add_executable(gui_host_tests tests/gui_host_tests.cpp tests/AllocationGuard.cpp)
  target_link_libraries(gui_host_tests PRIVATE clap openfilter_eq_engine openfilter_options PkgConfig::X11 PkgConfig::CAIRO PkgConfig::FONTCONFIG ${CMAKE_DL_LIBS})
  add_test(NAME gui_host COMMAND gui_host_tests $<TARGET_FILE:OpenFilterEQ>)
  set_tests_properties(gui_host PROPERTIES SKIP_RETURN_CODE 77)
endif()
